cmake_minimum_required(VERSION 3.22)
project(amdesktop LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(AM_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
if(NOT MSVC)
    add_compile_options("-ffile-prefix-map=${AM_ROOT}/=")
endif()
set(SRC "${AM_ROOT}/shared/cpp")
if(DEFINED ENV{SKIA_DIR})
    set(SKIA "$ENV{SKIA_DIR}" CACHE PATH "Skia source")
else()
    set(SKIA "${AM_ROOT}/third_party/skia" CACHE PATH "Skia source")
endif()
set(OUT "${AM_ROOT}/pc/build/native")
if(AM_ARCH)
    set(OUT "${OUT}-${AM_ARCH}")
endif()
set(ENGINE audio_dsp audio_fx_jni camera depth drawing easing effects interpolation model motion particles pieces3d preview scene shapes styles text textanim warp xml)
list(TRANSFORM ENGINE PREPEND "${SRC}/")
list(TRANSFORM ENGINE APPEND ".cpp")
add_library(amdesktop SHARED ${ENGINE} desktop_jni.cpp)
target_compile_definitions(amdesktop PRIVATE NDEBUG SK_DISABLE_TRACING XML_STATIC U_DISABLE_RENAMING=1 U_STATIC_IMPLEMENTATION U_USING_ICU_NAMESPACE=0)
target_include_directories(amdesktop PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${SRC}" "${SKIA}" "${SKIA}/third_party/externals/expat/expat/lib" "${SKIA}/third_party/externals/icu/source/common" "$ENV{JAVA_HOME}/include")
set_target_properties(amdesktop PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${OUT}" RUNTIME_OUTPUT_DIRECTORY "${OUT}")
foreach(config IN ITEMS RELEASE DEBUG RELWITHDEBINFO MINSIZEREL)
    set_target_properties(amdesktop PROPERTIES LIBRARY_OUTPUT_DIRECTORY_${config} "${OUT}" RUNTIME_OUTPUT_DIRECTORY_${config} "${OUT}")
endforeach()
if(APPLE)
    include("${AM_ROOT}/macos/native/macos.cmake")
elseif(WIN32)
    include("${AM_ROOT}/windows/native/windows.cmake")
else()
    include("${AM_ROOT}/linux/native/linux.cmake")
endif()

set(WHISPER_DIR "${AM_ROOT}/third_party/whisper.cpp" CACHE PATH "whisper.cpp source")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_subdirectory("${WHISPER_DIR}" whisper EXCLUDE_FROM_ALL)
target_sources(amdesktop PRIVATE "${SRC}/tracker/tracker.cpp" "${SRC}/tracker_jni.cpp" "${SRC}/whisper_jni.cpp" "${SRC}/absent_bg_jni.cpp" "${SRC}/absent_bg/absent_bg.cpp")
target_link_libraries(amdesktop PRIVATE whisper)
if(MSVC)
    set_source_files_properties("${SRC}/tracker/tracker.cpp" "${SRC}/tracker_jni.cpp" "${SRC}/whisper_jni.cpp" "${SRC}/absent_bg_jni.cpp" "${SRC}/absent_bg/absent_bg.cpp" PROPERTIES COMPILE_OPTIONS "/EHsc")
else()
    set_source_files_properties("${SRC}/tracker/tracker.cpp" "${SRC}/tracker_jni.cpp" "${SRC}/whisper_jni.cpp" "${SRC}/absent_bg_jni.cpp" "${SRC}/absent_bg/absent_bg.cpp" PROPERTIES COMPILE_OPTIONS "-fexceptions")
endif()

set(RNNOISE_DIR "${AM_ROOT}/third_party/rnnoise")
add_library(rnnoise STATIC
    "${RNNOISE_DIR}/src/denoise.c" "${RNNOISE_DIR}/src/rnn.c" "${RNNOISE_DIR}/src/rnn_data.c"
    "${RNNOISE_DIR}/src/pitch.c" "${RNNOISE_DIR}/src/kiss_fft.c" "${RNNOISE_DIR}/src/celt_lpc.c")
target_include_directories(rnnoise PUBLIC "${RNNOISE_DIR}/include" PRIVATE "${RNNOISE_DIR}/src")
target_compile_definitions(rnnoise PRIVATE RNNOISE_BUILD _USE_MATH_DEFINES)
target_sources(amdesktop PRIVATE "${SRC}/denoise_jni.cpp")
target_link_libraries(amdesktop PRIVATE rnnoise)

set(NCNN_DIR "${AM_ROOT}/third_party/ncnn" CACHE PATH "ncnn source")
foreach(option NCNN_VULKAN NCNN_SHARED_LIB NCNN_INSTALL_SDK NCNN_BUILD_TOOLS NCNN_BUILD_EXAMPLES NCNN_BUILD_BENCHMARK NCNN_BUILD_TESTS NCNN_PYTHON NCNN_C_API NCNN_INT8 NCNN_BF16)
    set(${option} OFF CACHE BOOL "" FORCE)
endforeach()
if(MSVC)
    set(NCNN_OPENMP OFF CACHE BOOL "" FORCE)
    set(NCNN_SIMPLEOMP OFF CACHE BOOL "" FORCE)
else()
    set(NCNN_OPENMP ON CACHE BOOL "" FORCE)
    set(NCNN_SIMPLEOMP ON CACHE BOOL "" FORCE)
endif()
set(NCNN_LAYERS input split concat convolution convolutiondepthwise deconvolution crop padding pooling interp memorydata pixelshuffle relu prelu sigmoid slice eltwise binaryop packing cast gemm reshape flatten noop innerproduct layernorm multiheadattention permute softmax)
file(STRINGS "${NCNN_DIR}/src/CMakeLists.txt" NCNN_LAYER_LINES REGEX "^ncnn_add_layer\\(")
foreach(line ${NCNN_LAYER_LINES})
    string(REGEX REPLACE "^ncnn_add_layer\\(([A-Za-z0-9]+).*" "\\1" layer "${line}")
    string(TOLOWER "${layer}" layer)
    if(layer IN_LIST NCNN_LAYERS)
        set(WITH_LAYER_${layer} ON CACHE BOOL "" FORCE)
    else()
        set(WITH_LAYER_${layer} OFF CACHE BOOL "" FORCE)
    endif()
endforeach()
add_subdirectory("${NCNN_DIR}" ncnn EXCLUDE_FROM_ALL)
set(RIFE_SOURCE "${AM_ROOT}/third_party/rife_warp/warp.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${RIFE_SOURCE}")
file(READ "${RIFE_SOURCE}" RIFE_TEXT)
string(FIND "${RIFE_TEXT}" "int Warp::forward(const std::vector<Mat>&" RIFE_START)
string(FIND "${RIFE_TEXT}" "int Warp::forward(const std::vector<VkMat>&" RIFE_END)
if(RIFE_START LESS 0 OR RIFE_END LESS_EQUAL RIFE_START)
    message(FATAL_ERROR "Cannot find the RIFE CPU warp")
endif()
math(EXPR RIFE_LENGTH "${RIFE_END} - ${RIFE_START}")
string(SUBSTRING "${RIFE_TEXT}" ${RIFE_START} ${RIFE_LENGTH} RIFE_CPU)
set(RIFE_CPU_FILE "${CMAKE_CURRENT_BINARY_DIR}/rife_cpu.cpp")
set(RIFE_CPU_TEXT "#include \"rife_cpu.h\"\n#include <algorithm>\n#include <cmath>\nusing namespace ncnn;\nWarp::Warp() { one_blob_only = false; }\n${RIFE_CPU}")
set(RIFE_CPU_OLD "")
if(EXISTS "${RIFE_CPU_FILE}")
    file(READ "${RIFE_CPU_FILE}" RIFE_CPU_OLD)
endif()
if(NOT RIFE_CPU_OLD STREQUAL RIFE_CPU_TEXT)
    file(WRITE "${RIFE_CPU_FILE}" "${RIFE_CPU_TEXT}")
endif()
target_sources(amdesktop PRIVATE ai_desktop.cpp "${SRC}/cutout.cpp" "${RIFE_CPU_FILE}")
target_compile_definitions(amdesktop PRIVATE AM_DESKTOP)
target_link_libraries(amdesktop PRIVATE ncnn)
