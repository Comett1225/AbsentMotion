cmake_minimum_required(VERSION 3.22)
project(amdesktop LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set(SRC "${CMAKE_CURRENT_SOURCE_DIR}/../../app/src/main/cpp")
set(SKIA "$ENV{SKIA_DIR}" CACHE PATH "Skia source")
set(SKIA_LIB_DIR "${SKIA}/out/win-x64" CACHE PATH "Skia libraries")
set(ENGINE audio_dsp audio_fx_jni camera drawing easing effects model motion particles pieces3d preview scene shapes styles text textanim warp xml)
foreach(name IN LISTS ENGINE)
    list(APPEND SOURCES "${SRC}/${name}.cpp")
endforeach()
add_library(amdesktop SHARED ${SOURCES} desktop_jni.cpp gpu_vulkan.cpp)
target_compile_definitions(amdesktop PRIVATE NDEBUG SK_DISABLE_TRACING XML_STATIC U_DISABLE_RENAMING=1 U_STATIC_IMPLEMENTATION U_USING_ICU_NAMESPACE=0 _USE_MATH_DEFINES NOMINMAX)
target_compile_options(amdesktop PRIVATE /w /GR- /EHs-c- /utf-8)
target_include_directories(amdesktop PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/compat" "${SRC}" "${SKIA}" "${SKIA}/third_party/externals/expat/expat/lib" "${SKIA}/third_party/externals/icu/source/common" "${SKIA}/third_party/externals/vulkan-headers/include" "$ENV{JAVA_HOME}/include" "$ENV{JAVA_HOME}/include/win32")
file(GLOB SKIA_LIBS "${SKIA_LIB_DIR}/*.lib")
target_link_libraries(amdesktop PRIVATE ${SKIA_LIBS} dwrite d2d1 dxgi d3d11 windowscodecs ole32 oleaut32 uuid usp10 user32 gdi32 advapi32 shell32 shlwapi)
set_target_properties(amdesktop PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../build/native")
